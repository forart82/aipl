### Camera Projection & Elevation Frustum Culling (`std/gfx25d/camera25d.aipl`)

- **Barrel Import**: `import * from "std/gfx25d/gfx25d_mod"`
- **Unit Test**: `tests/functions/gfx25d/camera25d_test.aipl` | **Examples**: `examples/ex_259_continuous_world_25d.aipl` .. `examples/ex_263_isometric_8dir_rig_and_lighting_lab_window.aipl`

```aipl
module tests.functions.gfx25d.camera25d_doc_test
import { createCamera25D, project3DToIso, unprojectIsoTo3D, getVisibleWorldAABB, updateCamera25D } from "std/gfx25d/camera25d"
import { assert, assertApproxEq } from "std/prelude"

pub fn test_camera25d_roundtrip(): int:
  mut cam = createCamera25D(1280.0, 720.0, 1.5, 45.0, 30.0)
  cam = updateCamera25D(cam, 120.0, 80.0, 24.0, 0.016)
  let scr = project3DToIso(cam, 120.0, 80.0, 24.0)
  let worldPt = unprojectIsoTo3D(cam, scr.x, scr.y, 24.0)
  assertApproxEq(worldPt.x, 120.0, 0.01, "x roundtrip")
  assertApproxEq(worldPt.y, 80.0, 0.01, "y roundtrip")
  let bounds = getVisibleWorldAABB(cam, -50.0, 150.0)
  assert(bounds.maxX > bounds.minX, "valid frustum AABB")
  ret 0
```

### Gouraud Meshes, Spline Ribbons, Lighting, Sprite Rigs & Weather (`std/gfx25d/gfx25d_mod.aipl`)

```aipl
module tests.functions.gfx25d.world_pipeline_doc_test
import { Vec3 } from "std/math/vec_math"
import { Color } from "std/gfx2d/shape2d"
import { shadeVertex25D, extrudeSplineRibbon25D, createPointLight25D, direction8FromVelocity, createIsoSpriteRig25D, stepIsoSpriteRig25D, createRainSystem25D, createEmberSmokeEmitter25D } from "std/gfx25d/gfx25d_mod"
import { assert, assertEqInt } from "std/prelude"

pub fn test_world_pipeline_25d(): int:
  let lit = shadeVertex25D(Color(r = 0.4, g = 0.7, b = 0.3, a = 1.0), Vec3(x = 0.0, y = 0.0, z = 1.0), Vec3(x = 0.0, y = 0.0, z = 1.0), 0.35)
  assert(lit.g > 0.3, "shaded green positive")
  let pts = [Vec3(x = 0.0, y = 0.0, z = 0.0), Vec3(x = 32.0, y = 16.0, z = 0.0), Vec3(x = 64.0, y = 32.0, z = 0.0)]
  let strip = extrudeSplineRibbon25D(pts, 8.0, 2.0)
  assertEqInt(strip.len(), 3, "3 ribbon vertices")
  let lamp = createPointLight25D(16.0, 16.0, 8.0, 96.0, Color(r = 1.0, g = 0.7, b = 0.3, a = 1.0))
  assert(lamp.radius > 0.0, "light radius positive")
  let dir = direction8FromVelocity(1.0, 0.0, 4)
  assert(dir >= 0 and dir < 8, "valid 8-way direction")
  mut rig = createIsoSpriteRig25D(0, 64, 64, 8, 6, 0.0, 0.0, 0.0)
  rig = stepIsoSpriteRig25D(rig, 12.0, 0.0, 0.0, 0.016)
  let rain = createRainSystem25D(24, 200.0)
  assertEqInt(rain.drops.len(), 24, "24 rain drops")
  let embers = createEmberSmokeEmitter25D(0.0, 0.0, 0.0, 12)
  assertEqInt(embers.particles.len(), 12, "12 ember particles")
  ret 0
```
